    periodic = T       ! Use periodical boundaries in x and z-direction
                       ! Only makes sense for vfuncnr=0 (zero velocity)
                       ! or vfuncnr=2 (shear flow).
    periodic_channel = F ! Use periodical boundaries in x-direction.
                       ! Only makes sense for vfuncnr=0 (zero velocity)
                       ! or vfuncnr=2 (shear flow).
    flowrate = F       ! Impose flowrate in x-direction (for
                       ! periodic=.true. or periodic_channel=.true. only)
    move_mesh_x = F    ! Move the mesh in x-direction to keep the center
                       ! of volume of the drop at the same position wrt
                       ! to the mesh
    mesh_plot = T      ! write mesh plot files/info (curves, mesh etc.)
    compute_error = F  ! Compute error. Only makes sense for eta_e=eta_i
                       ! (no viscosity difference) and gammac=0 (no
                       ! surface tension) or if vfuncnr=0 (no flow)
                       ! or vfuncnr=3 (rotation).
    compute_volume = T ! Compute volume and center of volume. 
    n=2                ! number of elements in x, y and z direction of box
    nc=20              ! number of elements on the equator of the sphere
    gauss = 4          ! order of integration for tetrahedral elements
    gaussb = 4         ! order of integration for triangular elements
    vfuncnr = 2        ! function number for the velocity field u
    choice_um = 2      ! choice for um in the motion of the interface:
                       !  0: um = 0
                       !  1: um = average velocity of all interface nodes
                       !  2: um = velocity of the center of volume 
    vtkevery = 10      ! vtk file every vtkevery steps. -1: means none
    ctime = T          ! continue time in output after restart
    restart = 0        ! restart=0: no restart
                       ! restart=1: normal restart
                       ! restart=2: restart, but start with Euler step
    restart_every = 100 ! write restart file every restart_every steps. 
    timeint = 2        ! time integration scheme
    numtimesteps = 200 ! number of time steps
    ox = -0.5          ! x-coordinate left lower corner
    oy = -0.5          ! y-coordinate left lower corner
    oz = -0.5          ! z-coordinate left lower corner
    lx = 1.0           ! length of the domain
    ly = 1.0           ! width of the domain
    lz = 1.0           ! height of the domain
    xc = 0.0 0.0 0.0   ! center of the initial sphere
    rc = 0.1           ! initial radius of the sphere
    eta_e = 1.0        ! external fluid viscosity
    eta_i = 1.0        ! internal fluid viscosity
    gammac = 0.5       ! surface tension coefficient
    factor = 3.      ! factor for the interface grid deformation
    U_avg  = 0.25      ! imposed average velocity for flowrate=.true.
    rs_up  = 1.4,      ! real_storage velocity-pressure LU HSL
    is_up  = 1.5,      ! integer_storage velocity-pressure LU HSL
    deltat = 3.e-2     ! time step
    beta = 0.5         ! factor beta in SUPG
